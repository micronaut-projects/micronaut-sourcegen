package test

import java.util.function.Function
import kotlin.Any
import kotlin.Int
import kotlin.String

public class Capturelocal {
  private var `receiver`: CapturedTarget = CapturedTarget("first")

  private var count: Int = 0

  public fun next(): CapturedTarget {
    this. count = this. count + 1
    return this. `receiver`
  }

  public fun reads(): Int {
    return this. count
  }

  public fun replace() {
    this. `receiver` = CapturedTarget("second")
  }

  public fun capture(): Function<Any, Any> {
    var target:CapturedTarget = this. `receiver`
    return target.let { target1 -> Function<Any, Any> { arg0 -> target1.apply(arg0 as String) } }
  }
}
