package test

import io.micronaut.sourcegen.CalleeBoundsRegressionTest
import io.micronaut.sourcegen.InvocationProgramRegressionTest
import java.lang.Runnable
import kotlin.Any
import kotlin.CharSequence
import kotlin.String

public class AfterCastintersection {
  public fun call(`value`: Any): String {
    CalleeBoundsRegressionTest.Calls.intersection(run {
      `value` as CharSequence
      `value` as Runnable
      `value`
    })
    return InvocationProgramRegressionTest.Calls.choose(`value`)
  }
}
